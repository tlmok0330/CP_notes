Matrix Fast Power : O(k^3 * log n)
{
const int MAT = 103;
struct M {
  int t[MAT][MAT];
  int n;
  M(int n) {
    this->n = n;
    memset(t, 0, sizeof(t));
  }
  M operator* (const M& a) const {
    M c = M(n);
    signed long long mo2 = (int)4 * MOD * MOD;
    for (int i = 0; i < n; i++) {
      for (int j = 0; j < n; j++ ) {
        for (int k = 0; k < n; k++) {
          c.t[i][k] += t[i][j] * a.t[j][k];
          if (c.t[i][k] > mo2) c.t[i][k] -= mo2;
        }
      }
    }
    for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) c.t[i][j] %= MOD;
    return c;
  }
};
M fastPow(int n, M base, int power) {
  M ret = M(n);
  for (int i = 0; i < n; i++) ret.t[i][i] = 1;
  while (power > 0) {
    if (power & 1) {
      ret = ret * base;
    }
    base = base * base;
    power >>= 1;
  }
  return ret;
}
}

DAG-Indegree: O(n)
{
  vector<int> in(n+1); vector<bool> del(n+1); vector<int> de;
  queue <int> Q;
  for (int i=1; i<=n; ++i) {
    for (auto v : edge[i]) {
      ++in[v];
    }
  }
  for (int i=1; i<=n; ++i) if (!in[i]) Q.push(i);
  while (!Q.empty()) {
    auto u = Q.front();
    de.push_back(u);
    Q.pop();
    del[u] = 1;
    for (auto v : edge[u]) {
      --in[v];
      if (!in[v]) {
        Q.push(v);
      }
    }
  }
}

MST KRUSKAL: O(E*logE)
{
  vector<vector<int> > se(103);
  vector<int> P(103), sz(103);
  //DSU
  int Find(int u){
    return (P[u]==u? u: P[u]=Find(P[u]));
  }
  void Union(int u, int v){
    int fu=Find(u), fv=Find(v);
    if(fu==fv) return;
    if(sz[fu]>sz[fv]) swap(fu, fv); //size of Find(u) is smaller
    sz[fv]+=sz[fu];
    
    se[fv].reserve(sz[fv] + sz[fu]);
    se[fv].insert(se[fv].end(), se[fu].begin(), se[fu].end());
    vector<int>().swap(se[fu]);   // Swaps with a temporary empty vector, freeing memory OR use list<int>
    
    P[fu]=fv;
    return;
  }
  
  int ans=0, co=0;
  for(int i=1; i<=n; i++) P[i]=i;
  sort(edge.begin(), edge.end());
  for(auto [we, fr, to]: edge){
    if(Find(fr)!=Find(to)){
      Union(fr, to);
      ans+=we;
      co++;
    }
  }
}

struct XorBasis {
    static constexpr int B = 60;
    long long b[B] = {};
    int rank = 0, total = 0;

    bool insert(long long x) {
        total++;
        for (int i = B - 1; i >= 0; i--) {
            if (!(x >> i & 1)) continue;
            if (!b[i]) { b[i] = x; rank++; return true; }
            x ^= b[i];
        }
        return false;
    }
    bool can_make(long long x) const {
        for (int i = B - 1; i >= 0; i--)
            if (x >> i & 1) { if (!b[i]) return false; x ^= b[i]; }
        return true;
    }
    long long max_xor(long long x = 0) const {
        for (int i = B - 1; i >= 0; i--) x = max(x, x ^ b[i]);
        return x;
    }
    // k-th smallest distinct value among all subset XORs, 0-indexed, empty subset (value 0) included.
    // Returns -1 if k >= 2^rank.
    long long kth(long long k) {
        for (int i = 0; i < B; i++) if (b[i])
            for (int j = i + 1; j < B; j++)
                if (b[j] >> i & 1) b[j] ^= b[i];
        vector<long long> v;
        for (int i = 0; i < B; i++) if (b[i]) v.push_back(b[i]);
        if (k >= (1LL << v.size())) return -1;
        long long res = 0;
        for (int i = 0; i < (int)v.size(); i++) if (k >> i & 1) res ^= v[i];
        return res;
    }
};

HLD + Segment Tree:
{
int dfs_size(int u, int p, int d) {
    int size = 1;
    int max_child_size = 0;
    parent[u] = p;
    depth[u] = d;
    heavy[u] = -1; // Default to no heavy child (leaves)

    for (int v : adj[u]) {
        if (v != p) {
            int child_size = dfs_size(v, u, d + 1);
            size += child_size;
            if (child_size > max_child_size) {
                max_child_size = child_size;
                heavy[u] = v; // Track child with max size
            }
        }
    }
    return size;
}

int cur_pos = 0;
void dfs_hld(int u, int h) {
    head[u] = h;
    pos[u] = ++cur_pos;
    // Map position back to initial node value for segment tree construction
    arr_flat[cur_pos] = val[u]; 

    // 1. Process the heavy child first to keep its sequence contiguous
    if (heavy[u] != -1) {
        dfs_hld(heavy[u], h); 
    }

    // 2. Process light children (each starts its own heavy path)
    for (int v : adj[u]) {
        if (v != parent[u] && v != heavy[u]) {
            dfs_hld(v, v); // The head of this new path is the child itself
        }
    }
}

int query_path(int a, int b) {
    int res = 0; // Assuming a sum query
    while (head[a] != head[b]) {
        // Move the node whose path head is deeper
        if (depth[head[a]] > depth[head[b]]) {
            swap(a, b);
        }
        // Query the range from the head of b's path up to b itself
        res += segment_tree_query(pos[head[b]], pos[b]);
        // Jump above the current heavy path
        b = parent[head[b]];
    }
    // Now both nodes are on the identical heavy path
    if (depth[a] > depth[b]) {
        swap(a, b);
    }
    // Final query on the remaining segment between them
    res += segment_tree_query(pos[a], pos[b]);
    return res;
}

}

ConvexHull: 
{
struct pt {
    double x, y;
    bool operator == (pt const& t) const {
        return x == t.x && y == t.y;
    }
    pt operator-(const pt& rhs) const {
        return {x - rhs.x, y - rhs.y};
    }
    // Cross product
    double operator*(const pt& rhs) const {
        return x * rhs.y - y * rhs.x;
    }
};

int orientation(pt a, pt b, pt c) {
    double v = a.x*(b.y-c.y)+b.x*(c.y-a.y)+c.x*(a.y-b.y);
    if (v < 0) return -1; // clockwise
    if (v > 0) return +1; // counter-clockwise
    return 0;
}

bool cw(pt a, pt b, pt c, bool include_collinear) {
    int o = orientation(a, b, c);
    return o < 0 || (include_collinear && o == 0);
}
bool collinear(pt a, pt b, pt c) { return orientation(a, b, c) == 0; }

int dist2(const pt &a, const pt &b) {
    int dx = a.x - b.x, dy = a.y - b.y;
    return dx*dx + dy*dy;
}

// Helper: copy array and sort as needed
void convex_hull(pt a[], int n, pt out[], int &m, bool include_collinear = false) {
    // Find lowest point
    int p0 = 0;
    for (int i = 1; i < n; ++i)
        if (a[i].y < a[p0].y || (a[i].y == a[p0].y && a[i].x < a[p0].x)) p0 = i;
    std::swap(a[0], a[p0]);

    // Sort rest by polar angle
    std::sort(a+1, a+n, [&](const pt &p, const pt &q) {
        int o = orientation(a[0], p, q);
        if (o == 0)
            return (dist2(a[0], p) < dist2(a[0], q)); // You might want your own dist2 function
            // FRAGILE? int vs double
        return o == 1;
    });

    // Graham scan
    int k = 0;
    for (int i = 0; i < n; ++i) {
        while (k >= 2 && orientation(out[k-2], out[k-1], a[i]) < (include_collinear ? 0 : 1))
            --k;
        out[k++] = a[i];
    }

    // Remove final duplicate if collinear not allowed
    if (!include_collinear && k >= 2 && out[0] == out[k-1]) --k;
    m = k;
}

int pointLocation(pt p, pt hull[], int n) {
    bool onBoundary = false;
    // int n=hull.size()-1;
    for(int i = 0; i < n; i++) {
        pt a = hull[i];
        pt b = hull[(i + 1) % n];
        
        double cross = (b - a) * (p - a);
        
        // Check if point is on the line segment
        if(abs(cross) < 1e-9) {
            // Check if point is between a and b
            if(min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
               min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y)) {
                onBoundary = true;
            }
        }
        
        // If cross product is negative, point is on wrong side
        if(cross < -1e-9) return -1; // Outside
    }
    
    return onBoundary ? 0 : 1; // On boundary or inside
}

    return (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
}
// Counter-clockwise hull, collinear points removed. For coordinates up to 1e9 use __int128 in cross.
vector<Pt> convex_hull(vector<Pt> p) {
    sort(p.begin(), p.end());
    p.erase(unique(p.begin(), p.end()), p.end());
    int n = p.size();
    if (n <= 2) return p;
    vector<Pt> h(2 * n);
    int k = 0;
    for (int i = 0; i < n; i++) {
        while (k >= 2 && cross(h[k - 2], h[k - 1], p[i]) <= 0) k--;
        h[k++] = p[i];
    }
    for (int i = n - 2, lo = k + 1; i >= 0; i--) {
        while (k >= lo && cross(h[k - 2], h[k - 1], p[i]) <= 0) k--;
        h[k++] = p[i];
    }
    h.resize(k - 1);
    return h;
}


struct MergeSortTree {
  int n;
  vector<vector<int>> tree;  

  MergeSortTree(const vector<int> &a) {
    n = (int)a.size();
    tree.assign(4 * n, {});
    build(1, 0, n - 1, a);
  }

  void build(int node, int l, int r, const vector<int> &a) {
    if (l == r) {
      tree[node] = {a[l]};
      return;
    }
    int mid = (l + r) >> 1;
    build(node << 1, l, mid, a);
    build(node << 1 | 1, mid + 1, r, a);
    tree[node].resize(tree[node << 1].size() + tree[node << 1 | 1].size());
    merge(tree[node << 1].begin(), tree[node << 1].end(),
          tree[node << 1 | 1].begin(), tree[node << 1 | 1].end(),
          tree[node].begin());
  }

  int query(int node, int l, int r, int ql, int qr, int k) const {
    if (qr < l || r < ql) return 0;             // no overlap
    if (ql <= l && r <= qr) {                   // fully inside
      return upper_bound(tree[node].begin(), tree[node].end(), k) - tree[node].begin();
    }
    int mid = (l + r) >> 1;
    return query(node << 1, l, mid, ql, qr, k) + query(node << 1 | 1, mid + 1, r, ql, qr, k);
  }

  int query(int l, int r, int k) const {
    return query(1, 0, n - 1, l, r, k);
  }
};

Hashing - vector of int -> ull - O(n)
{ 
static const ull SEED =
    chrono::steady_clock::now().time_since_epoch().count();

static ull splitmix64(ull x) {
    x += 0x9e3779b97f4a7c15ULL;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
    return x ^ (x >> 31);
}

ull compute_hash(const vector<int>& v) {
    ull h = SEED;
    for (int x : v)
        h = splitmix64(h ^ splitmix64((ull)(int64_t)x));
    return h;
}
}

Eppstein_k_shortest_paths: O(m+k+nlogn) [paths with cycles allowed]
{
#include <iostream>
#include <vector>
#include <queue>
#include <unordered_map>

// Reuses the identical Graph definition structures from snippet 1

std::vector<Path> eppstein_k_shortest_paths(const Graph& graph, Node source, Node target, int K) {
    std::vector<Path> paths_found;
    std::unordered_map<Node, int> visit_counts;

    // Queue parameters: Path entity wrappers
    std::priority_queue<Path, std::vector<Path>, std::greater<>> pq;
    pq.push({0.0, {source}});

    while (!pq.empty() && paths_found.size() < K) {
        Path current = pq.top();
        pq.pop();

        Node current_node = current.nodes.back();
        visit_counts[current_node]++;

        if (current_node == target) {
            paths_found.push_back(current);
        }

        // Branching check: Cut execution branches if a node is requested over K iterations
        if (visit_counts[current_node] > K) {
            continue;
        }

        if (graph.find(current_node) != graph.end()) {
            for (const auto& edge : graph.at(current_node)) {
                Path next_path = current;
                next_path.nodes.push_back(edge.to);
                next_path.cost += edge.weight;
                pq.push(next_path);
            }
        }
    }
    return paths_found;
}
}

T-join using bitmask DP:
{
const long long INF = 1e15; 
long long solveMinWeightTJoin(const vector<vector<long long>>& dist, const vector<int>& T) {
    int k = T.size();
    if (k % 2 != 0) return -1; // T 的數量必須是偶數

    vector<long long> dp(1 << k, INF);
    dp[0] = 0; 

    for (int mask = 0; mask < (1 << k); ++mask) {
        if (dp[mask] == INF) continue;
        int i = -1;
        for (int b = 0; b < k; ++b) {
            if (!(mask & (1 << b))) {
                i = b;
                break;
            }
        }
        if (i == -1) continue;

        for (int j = i + 1; j < k; ++j) {
            if (!(mask & (1 << j))) {
                int next_mask = mask | (1 << i) | (1 << j);
                dp[next_mask] = min(dp[next_mask], dp[mask] + dist[T[i]][T[j]]);
            }
        }
    }
    return dp[(1 << k) - 1];
}
}

Two-Satisfiability template:
- 0-indexed variables 
  1. Clause: (x0 is True) OR (x1 is False)
  solver.either(0, true, 1, false);
  2. Clause: (x1 is True) OR (x2 is True)
  solver.either(1, true, 2, true);
  3. Force x2 to be False
  solver.force(2, false);
- O(N+M) 

struct TwoSat {
    
    int n, timer = 0, ncomp = 0;
    vector<vector<int>> g;
    vector<int> comp, low, ord, stk;
    vector<bool> val;
    TwoSat(int n) : n(n), g(2 * n) {}
    // literal id: 2*x = "x is true", 2*x+1 = "x is false"
    // adds clause (x_a == va) OR (x_b == vb)
    void either(int a, bool va, int b, bool vb) {
        int A = 2 * a + !va, B = 2 * b + !vb;
        g[A ^ 1].push_back(B);
        g[B ^ 1].push_back(A);
    }
    void force(int a, bool va) { either(a, va, a, va); }
    void dfs(int u) {
        ord[u] = low[u] = ++timer;
        stk.push_back(u);
        for (int v : g[u]) {
            if (!ord[v]) { dfs(v); low[u] = min(low[u], low[v]); }
            else if (comp[v] < 0) low[u] = min(low[u], ord[v]);
        }
        if (low[u] == ord[u]) {
            while (true) {
                int v = stk.back(); stk.pop_back();
                comp[v] = ncomp;
                if (v == u) break;
            }
            ncomp++;
        }
    }
    bool solve() {
        ord.assign(2 * n, 0); low.assign(2 * n, 0); comp.assign(2 * n, -1);
        for (int i = 0; i < 2 * n; i++) if (!ord[i]) dfs(i);
        val.assign(n, false);
        for (int i = 0; i < n; i++) {
            if (comp[2 * i] == comp[2 * i + 1]) return false;
            val[i] = comp[2 * i] < comp[2 * i + 1];  // Tarjan numbers SCCs in reverse topological order
        }
        return true;
    }
};
